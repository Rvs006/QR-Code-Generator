function hashStr(s) {
  let h = 0;
  for (let i = 0; i < s.length; i++) { h = ((h << 5) - h + s.charCodeAt(i)) | 0; }
  return Math.abs(h);
}

function PseudoQR({ payload, size, className }) {
  const canvasRef = useRef(null);
  const sz = size || 120;
  const modules = 29;

  const drawQR = useCallback(() => {
    const cv = canvasRef.current;
    if (!cv) return;
    const ctx = cv.getContext('2d');
    const m = sz / modules;
    ctx.fillStyle = '#ffffff';
    ctx.fillRect(0, 0, sz, sz);
    ctx.fillStyle = '#000000';

    const seed = hashStr(payload || 'empty');
    const grid = Array.from({ length: modules }, () => Array(modules).fill(false));

    const drawFinder = (ox, oy) => {
      for (let r = 0; r < 7; r++) for (let cc = 0; cc < 7; cc++) {
        const edge = r === 0 || r === 6 || cc === 0 || cc === 6;
        const inner = r >= 2 && r <= 4 && cc >= 2 && cc <= 4;
        if (edge || inner) grid[oy + r][ox + cc] = true;
      }
    };

    drawFinder(0, 0);
    drawFinder(modules - 7, 0);
    drawFinder(0, modules - 7);

    for (let i = 8; i < modules - 8; i++) {
      if (i % 2 === 0) { grid[6][i] = true; grid[i][6] = true; }
    }

    const ax = 20; const ay = 20;
    for (let r = -2; r <= 2; r++) for (let cc = -2; cc <= 2; cc++) {
      const edge = Math.abs(r) === 2 || Math.abs(cc) === 2;
      const center = r === 0 && cc === 0;
      if (edge || center) grid[ay + r][ax + cc] = true;
    }

    let s = seed;
    for (let r = 0; r < modules; r++) for (let cc = 0; cc < modules; cc++) {
      if (grid[r][cc]) continue;
      if (r < 9 && cc < 9) continue;
      if (r < 9 && cc >= modules - 8) continue;
      if (r >= modules - 8 && cc < 9) continue;
      if (r === 6 || cc === 6) continue;
      if (r >= ay - 2 && r <= ay + 2 && cc >= ax - 2 && cc <= ax + 2) continue;
      s = ((s * 1103515245 + 12345) & 0x7fffffff);
      if (s % 100 < 48) grid[r][cc] = true;
    }

    for (let r = 0; r < modules; r++) for (let cc = 0; cc < modules; cc++) {
      if (grid[r][cc]) ctx.fillRect(Math.floor(cc * m), Math.floor(r * m), Math.ceil(m), Math.ceil(m));
    }
  }, [payload, sz]);

  useEffect(() => { drawQR(); }, [drawQR]);

  return <canvas ref={canvasRef} width={sz} height={sz} className={className} />;
}

function validateRows(data) {
  const tagCount = {};
  data.forEach(r => { tagCount[r.assetTag] = (tagCount[r.assetTag] || 0) + 1; });
  return data.map((r, i) => {
    if (!r.payload || r.payload.trim() === '') return { ...r, valid: false, warning: 'Empty payload — QR will have no data' };
    let isJson = false;
    try { JSON.parse(r.payload); isJson = true; } catch(e) {}
    if (!isJson) return { ...r, valid: true, warning: 'Payload is not valid JSON' };
    if (tagCount[r.assetTag] > 1) return { ...r, valid: true, warning: `Duplicate asset tag — appears ${tagCount[r.assetTag]} times` };
    if (r.payload.length > 150) return { ...r, valid: true, warning: `Long payload (${r.payload.length} chars) — may need QR Version 7+` };
    return { ...r, valid: true, warning: null };
  });
}

                <div className="flex items-center justify-between mb-1.5">
                  <span className="text-[12px]" style={{ color: c.tx2 }}>Scanning {verifyResults.length} of {galleryItems.length}...</span>
                  <span className="text-[12px] font-semibold" style={{ color: '#00B0F0', fontFamily: "'JetBrains Mono', monospace" }}>{verifyProgress}%</span>
                </div>
                <div className="w-full h-1.5 rounded-full overflow-hidden" style={{ background: c.bg3 }}>
                  <div className="h-full rounded-full transition-all duration-150" style={{ background: 'linear-gradient(90deg, #00B0F0, #4CAF50)', width: `${verifyProgress}%` }} />
                </div>
              </div>
            )}

            {!verifyRunning && verifyResults.length > 0 && (
              <div className="px-6 py-3 flex items-center gap-4 flex-shrink-0" style={{ borderBottom: `1px solid ${c.bdr}` }}>
                <div className="flex items-center gap-1.5">
                  <CheckCircle className="w-4 h-4 text-[#4CAF50]" />
                  <span className="text-[13px] text-[#4CAF50] font-bold" style={{ fontFamily: "'JetBrains Mono', monospace" }}>{verifyResults.filter(r => r.passed).length}</span>
                  <span className="text-[13px]" style={{ color: c.tx3 }}>passed</span>
                </div>
                {verifyResults.some(r => !r.passed) && (
                  <div className="flex items-center gap-1.5">
                    <XCircle className="w-4 h-4 text-[#E53935]" />
                    <span className="text-[13px] text-[#E53935] font-bold" style={{ fontFamily: "'JetBrains Mono', monospace" }}>{verifyResults.filter(r => !r.passed).length}</span>
                    <span className="text-[13px]" style={{ color: c.tx3 }}>failed</span>
                  </div>
                )}
                {verifyResults.every(r => r.passed) && (
                  <div className="flex items-center gap-1.5 ml-2 px-2.5 py-1 rounded-lg" style={{ background: 'rgba(76,175,80,0.1)' }}>
                    <Shield className="w-3.5 h-3.5 text-[#4CAF50]" />
                    <span className="text-[12px] text-[#4CAF50] font-semibold">All codes verified — safe to print</span>
                  </div>
                )}
              </div>
            )}

            <div className="flex-1 overflow-auto px-6 py-3">
              <table className="w-full text-[13px]">
                <thead>
                  <tr className="sticky top-0 z-10" style={{ background: c.bg1 }}>
                    <th className="px-2 py-2 text-left text-[11px] font-semibold uppercase tracking-wider w-10" style={{ color: c.tx3 }}>Status</th>
                    <th className="px-2 py-2 text-left text-[11px] font-semibold uppercase tracking-wider" style={{ color: c.tx3 }}>Asset Tag</th>
                    <th className="px-2 py-2 text-left text-[11px] font-semibold uppercase tracking-wider" style={{ color: c.tx3 }}>Folder</th>
                    <th className="px-2 py-2 text-left text-[11px] font-semibold uppercase tracking-wider" style={{ color: c.tx3 }}>Result</th>
                  </tr>
                </thead>
                <tbody>
                  {verifyResults.map((r, i) => (
                    <tr key={i} style={{ borderTop: `1px solid ${c.bdr}` }}>
                      <td className="px-2 py-2">{r.passed ? <CheckCircle className="w-4 h-4 text-[#4CAF50]" /> : <XCircle className="w-4 h-4 text-[#E53935]" />}</td>
                      <td className="px-2 py-2 font-semibold" style={{ fontFamily: "'JetBrains Mono', monospace" }}>{r.assetTag}</td>
                      <td className="px-2 py-2" style={{ color: c.tx2 }}>{r.folder}</td>
                      <td className="px-2 py-2 text-[12px]" style={{ color: r.passed ? '#4CAF50' : '#E53935' }}>{r.detail}</td>
                    </tr>
                  ))}
                </tbody>
              </table>
              {verifyRunning && verifyResults.length === 0 && (
                <div className="py-12 text-center">
                  <ScanLine className="w-8 h-8 mx-auto mb-3" style={{ color: c.tx3 }} />
                  <p className="text-[13px]" style={{ color: c.tx3 }}>Starting scan verification...</p>
                </div>
              )}
            </div>

            <div className="px-6 py-3 flex justify-end flex-shrink-0" style={{ borderTop: `1px solid ${c.bdr}` }}>
              <button className="px-4 py-2 rounded-lg text-[13px] font-semibold transition-colors" style={{ color: c.tx2, border: `1px solid ${c.bdr}` }} onClick={handleCloseVerify}>{verifyRunning ? 'Stop & Close' : 'Close'}</button>
            </div>
          </div>
        </div>
      )}

      {/* FOLDER TREE / DOWNLOAD MODAL */}
      {showFolderTree && (
        <div className="fixed inset-0 z-[200] flex items-center justify-center">
          <div className="absolute inset-0 backdrop-blur-sm" style={{ background: d ? 'rgba(0,0,0,0.6)' : 'rgba(0,0,0,0.3)' }} onClick={() => setShowFolderTree(false)} />
          <div className="relative rounded-2xl max-w-lg w-[90%] max-h-[80vh] overflow-auto p-6 shadow-2xl" style={{ background: c.bg1, border: `1px solid ${c.bdr}` }}>
            <div className="flex items-center justify-between mb-4">
              <h3 className="text-[16px] font-bold flex items-center gap-2"><FolderOpen className="w-5 h-5 text-[#F5A623]" />ZIP Structure Preview</h3>
              <button className="p-1" style={{ color: c.tx3 }} onClick={() => setShowFolderTree(false)}><X className="w-4 h-4" /></button>
            </div>
            <p className="text-[13px] mb-4" style={{ color: c.tx2 }}>QR images will be organized into this folder structure:</p>

            <div className="rounded-xl p-4 mb-4" style={{ background: c.bg, border: `1px solid ${c.bdr}`, fontFamily: "'JetBrains Mono', monospace", fontSize: '13px' }}>
              <div className="flex items-center gap-2 mb-2" style={{ color: c.tx }}>
                <FolderOpen className="w-4 h-4 text-[#F5A623]" />
                <span className="font-bold">asset_labels_v3/</span>
              </div>
              {Object.entries(folderTree).map(([mainFolder, subs]) => (
                <div key={mainFolder} className="ml-5">
                  <div className="flex items-center gap-2 py-1" style={{ color: c.tx }}>
                    <div className="w-px h-3" style={{ background: c.bdr, marginRight: '2px' }} />
                    <Folder className="w-3.5 h-3.5 text-[#2A5A9E]" />
                    <span className="font-semibold">{mainFolder}/</span>
                  </div>
                  {Object.entries(subs).map(([subFolder, assets]) => (
                    <div key={subFolder || '_root'} className="ml-5">
                      {subFolder && (
                        <div className="flex items-center gap-2 py-1" style={{ color: c.tx2 }}>
                          <div className="w-px h-3" style={{ background: c.bdr, marginRight: '2px' }} />
                          <Folder className="w-3.5 h-3.5 text-[#00B0F0]" />
                          <span>{subFolder}/</span>
                        </div>
                      )}
                      <div className={subFolder ? 'ml-5' : ''}>
                        {assets.map((tag, ai) => (
                          <div key={ai} className="flex items-center gap-2 py-0.5" style={{ color: c.tx3 }}>
                            <div className="w-px h-3" style={{ background: c.bdr, marginRight: '2px' }} />
                            <File className="w-3 h-3" style={{ color: c.tx3 }} />
                            <span className="text-[12px]">{tag}.{config.format}</span>
                          </div>
                        ))}
                      </div>
                    </div>
                  ))}
                </div>
              ))}
            </div>

            <div className="flex items-center gap-3 mb-4 px-1">
              <span className="text-[12px]" style={{ color: c.tx3 }}>{galleryItems.length} files · {Object.keys(folderTree).length} top-level folders · {config.format.toUpperCase()} format</span>
            </div>

            <div className="flex justify-end gap-2">
              <button className="px-4 py-2 rounded-lg text-[13px] transition-colors" style={{ color: c.tx2, border: `1px solid ${c.bdr}` }} onClick={() => setShowFolderTree(false)}>Cancel</button>
              <button className="flex items-center gap-2 bg-[#4CAF50] text-white px-5 py-2 rounded-lg font-semibold text-[13px] transition-colors" onMouseEnter={(e) => e.currentTarget.style.background = '#388E3C'} onMouseLeave={(e) => e.currentTarget.style.background = '#4CAF50'} onClick={() => setShowFolderTree(false)}><Download className="w-4 h-4" />Download ZIP</button>
            </div>
          </div>
        </div>
      )}

      {/* Click-away handlers */}
      {(showPresetDropdown || showRecentFiles) && <div className="fixed inset-0 z-40" onClick={() => { setShowPresetDropdown(false); setShowRecentFiles(false); }} />}
    </div>
  );
}

export default App;
